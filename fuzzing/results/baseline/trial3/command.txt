# Trial 3 - cJSON 1.7.19 AFL++ baseline replicate (branch trial3 @ baseline tag e8c1bfa)
# Host: WSL2 Ubuntu 24.04 x86_64, repo at ~/cJSON (ext4). Run all commands from repo root.

# --- Build (afl-clang-fast + AFL_USE_ASAN=1; ENABLE_SANITIZERS also adds ASan/UBSan) ---
rm -rf fuzzing/afl-build && mkdir -p fuzzing/afl-build && cd fuzzing/afl-build
CC=afl-clang-fast AFL_USE_ASAN=1 cmake ../.. -DENABLE_FUZZING=On -DENABLE_SANITIZERS=On -DBUILD_SHARED_LIBS=Off
AFL_USE_ASAN=1 make afl-main
cd ../..

# --- One-time host prep (WSL pipes cores to wsl-capture-crash; afl-fuzz refuses that) ---
echo core | sudo tee /proc/sys/kernel/core_pattern

# --- Coverage logger (5-minute interval) ---
bash fuzzing/log_coverage.sh fuzzing/results/baseline/trial3/findings/default/fuzzer_stats fuzzing/results/baseline/trial3/coverage.csv &
LOGGER_PID=$!

# --- Campaign (3600s) ---
AFL_SKIP_CPUFREQ=1 timeout -s INT 3600s afl-fuzz -i fuzzing/results/baseline/trial3/initial_corpus -o fuzzing/results/baseline/trial3/findings -x fuzzing/json.dict -- fuzzing/afl-build/fuzzing/afl-main @@ yes

# --- Teardown ---
kill "$LOGGER_PID"
cp fuzzing/results/baseline/trial3/findings/default/fuzzer_stats fuzzing/results/baseline/trial3/final_fuzzer_stats.txt
